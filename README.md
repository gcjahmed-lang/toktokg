function calcDist(lat1,lng1,lat2,lng2){let R=6371;let dLat=(lat2-lat1)*Math.PI/180;let dLng=(lng2-lng1)*Math.PI/180;let a=Math.sin(dLat/2)**2+Math.cos(lat1*Math.PI/180)*Math.cos(lat2*Math.PI/180)*Math.sin(dLng/2)**2;return (2*R*Math.asin(Math.sqrt(a))).toFixed(1);}
